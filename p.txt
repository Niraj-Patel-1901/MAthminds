// templates/js/partial-diff.js

document.addEventListener('DOMContentLoaded', () => {
    const solveBtn = document.getElementById('solveBtn');
    const resetBtn = document.getElementById('resetBtn');
    const problemTypeSelect = document.getElementById('problemType');
    const functionInput = document.getElementById('function');
    const var1Input = document.getElementById('var1');
    const var2Input = document.getElementById('var2');
    const var3Input = document.getElementById('var3');
    const orderInput = document.getElementById('order');
    const solutionDiv = document.getElementById('solution');

    // Show/hide fields depending on problem type
  problemTypeSelect.addEventListener('change', () => {
    const type = problemTypeSelect.value;

    // Always show var1 and var2
    var1Input.style.display = 'block';
    var2Input.style.display = 'block';

    // Show var3 only for Total Differential
    if (type === 'total') {
        var3Input.style.display = 'block';
    } else {
        var3Input.style.display = 'none';
        var3Input.value = '';
    }

    // Show order only for Partial Derivatives
    if (type === 'partial') {
        orderInput.parentElement.style.display = 'block';
    } else {
        orderInput.parentElement.style.display = 'none';
        orderInput.value = '';
    }
});

    solveBtn.addEventListener('click', async () => {
        const problemType = problemTypeSelect.value;
        const func = functionInput.value.trim();
        const var1 = var1Input.value.trim();
        const var2 = var2Input.value.trim();
        const var3 = var3Input.value.trim();
        const order = orderInput.value.trim();

        if (!func) {
            solutionDiv.innerHTML = `<p class="text-red-500">Please enter a function.</p>`;
            return;
        }

        let payload = { function: func };

        if (problemType === 'partial') {
            payload.mode = order && order > 1 ? "higher" : "first";
            if (payload.mode === "first") {
                payload.var = var1 || 'x';
            } else {
                payload.sequence = Array(parseInt(order) || 2).fill(var1 || 'x');
            }
        } 
        else if (problemType === 'chain') {
            payload.mode = "composite";
            payload.subs = { u: var1 || 'x', v: var2 || 'y' };
            payload.wrt = var3 || 'x';
        } 
        else if (problemType === 'total') {
            payload.mode = "composite"; // or "total" if you add it later
            payload.subs = { u: var1 || 'x', v: var2 || 'y', w: var3 || 'z' };
            payload.wrt = 'x';
        } 
        else if (problemType === 'euler') {
            payload.mode = "euler";
            if (order) payload.degree = parseInt(order);
        }

        try {
            const res = await fetch('/api/partial_diff', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const data = await res.json();

            if (!data.success) throw new Error(data.error);

            let html = data.steps.map(s => `<p>${s}</p>`).join("");
            if (data.table) {
                html += "<table class='table-auto border-collapse border border-gray-300 mt-4'><thead><tr>" +
                    data.table.headers.map(h => `<th class='border border-gray-300 px-2 py-1'>${h}</th>`).join("") +
                    "</tr></thead><tbody>" +
                    data.table.rows.map(r => `<tr>${r.map(c => `<td class='border border-gray-300 px-2 py-1'>\\(${c}\\)</td>`).join("")}</tr>`).join("") +
                    "</tbody></table>";
            }
            solutionDiv.innerHTML = html;
            MathJax.typesetPromise();
        } catch (err) {
            solutionDiv.innerHTML = `<p class="text-red-500">${err.message}</p>`;
        }
    });

    resetBtn.addEventListener('click', () => {
        problemTypeSelect.value = 'partial';
        functionInput.value = '';
        var1Input.value = '';
        var2Input.value = '';
        var3Input.value = '';
        orderInput.value = '';
        solutionDiv.innerHTML = '<p class="text-gray-700">Enter a function to find its partial derivatives.</p>';
        problemTypeSelect.dispatchEvent(new Event('change'));
    });

    // Initialize state
    problemTypeSelect.dispatchEvent(new Event('change'));
});
document.addEventListener('DOMContentLoaded', function() {
    const toggleBtn = document.getElementById('toggleFormula');
    const content = document.getElementById('formulaContent');
    const icon = document.getElementById('dropdownIcon');
    toggleBtn.addEventListener('click', function() {
        content.classList.toggle('hidden');
        icon.classList.toggle('rotate-180');
        if (!content.classList.contains('hidden')) {
            if (window.MathJax) MathJax.typesetPromise();
        }
    });
});
